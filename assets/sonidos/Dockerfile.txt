# Laboratorio CTF - Escalada de Privilegios con SUID
# Imagen base: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Evitar prompts interactivos durante build
ENV DEBIAN_FRONTEND=noninteractive

# Etiquetas para identificar el propósito educativo
LABEL description="Laboratorio CTF - Reto de Escalada de Privilegios SUID"
LABEL version="1.0"
LABEL maintainer="Laboratorio de Seguridad"

# Actualizar repositorios e instalar paquetes esenciales
RUN apt-get update && apt-get install -y \
    curl \
    wget \
    vim \
    net-tools \
    python3 \
    sudo \
    openssh-server \
    && rm -rf /var/lib/apt/lists/*

# ===============================================
# CONFIGURACIÓN SSH
# ===============================================
# Crear directorio /run/sshd requerido por SSH
RUN mkdir -p /run/sshd

# Configurar SSH para permitir login con contraseña
RUN sed -i 's/#PermitRootLogin prohibit-password/PermitRootLogin no/' /etc/ssh/sshd_config && \
    sed -i 's/#PasswordAuthentication yes/PasswordAuthentication yes/' /etc/ssh/sshd_config

# ===============================================
# USUARIO REGULAR: 'estudiante'
# ===============================================
# Crear usuario regular SIN sudo (usuario limitado)
RUN useradd -m -s /bin/bash estudiante && \
    echo 'estudiante:alumno123' | chpasswd

# ===============================================
# ARCHIVO SECRETO / FLAG
# ===============================================
# Crear archivo flag.txt en /root con contenido
# Permisos: 600 (solo lectura por root)
RUN echo "CTF{SUID_FIND_EXPLOTADO}" > /root/flag.txt && \
    chmod 600 /root/flag.txt && \
    chown root:root /root/flag.txt

# ===============================================
# CONFIGURACIÓN SUID VULNERABLE
# ===============================================
# Establecer bit SUID en /usr/bin/find
# Esto permite que cualquier usuario ejecute find con privilegios de root
RUN chmod u+s /usr/bin/find

# ===============================================
# MANTENIMIENTO DEL CONTENEDOR
# ===============================================
# Configurar directorio de trabajo
WORKDIR /home/estudiante

# Exponer puerto SSH del contenedor
EXPOSE 22

# Mantener el contenedor corriendo con SSH activo en foreground
CMD ["/usr/sbin/sshd", "-D", "-e"]
